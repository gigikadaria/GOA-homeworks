/* color - განსაზღვრავს ტექსტის ფერს */ p { color: red; /* აქ ტექსტი იქნება წითელი */ } /* HEX ფერის კოდი */ /* HEX ფორმატი იწერება ასე: #RRGGBB */ /* RR = წითელი, GG = მწვანე, BB = ლურჯი */ /* მნიშვნელობები არის 00-დან FF-მდე */ h1 { color: #0000FF; /* ლურჯი ფერი hex კოდით */ } /* RGB ფერის სისტემა */ /* rgb ნიშნავს Red, Green, Blue */ /* მნიშვნელობები იწერება 0-დან 255-მდე */ h2 { color: rgb(0, 255, 0); /* მწვანე ფერი rgb ფორმატით */ } /* background-color - აძლევს ელემენტს ფონის ფერს */ body { background-color: lightgray; /* გვერდის ფონი გახდება ღია ნაცრისფერი */ }
